Finite State Machine Implementation Plan for runepkg
The core requirement of a robust transactional package manager is ensuring atomic state transitions: if any step of an installation, upgrade, or removal pipeline fails halfway through, the system must abort safely, roll back partial file writes, and clean up temporary states without leaving the host system or the runepkg_db in a corrupted, half-baked condition.

Here is a structured implementation plan to integrate a robust Finite State Machine (FSM) transactional lifecycle into runepkg.

1. Define the FSM Transactional States
Implement an explicit state enum in the low-level C89 core and extend status tracking into the C++ high-level engine.

typedef enum {
    RUNEPKG_STATE_IDLE = 0,
    RUNEPKG_STATE_PREPARING,   /* Resolving dependencies, verifying signatures, acquiring locks */
    RUNEPKG_STATE_FETCHING,    /* Downloading .debs or unpacking source runes */
    RUNEPKG_STATE_VALIDATING,  /* SHA256/MD5 hashing, GPG InRelease check */
    RUNEPKG_STATE_STAGING,     /* Extracting payload to temporary staging sysroot */
    RUNEPKG_STATE_COMMITTING,  /* Atomic rename/swap into active filesystem & FNV-1a DB update */
    RUNEPKG_STATE_ROLLBACK,    /* Error handler: restoring previous state from journal */
    RUNEPKG_STATE_CLEANUP,     /* Purging temporary build/download directories */
    RUNEPKG_STATE_FAILED       /* Terminal abort state */
} RunepkgState;

2. Transaction Journaling & Rollback Architecture
To guarantee that a failed command leaves "nothing happening" to the active system, implement a journaling layer inside /var/lib/runepkg_dir/transaction.lock:

Pre-Execution Snapshot: Before mutating any files under install_dir (default /) or writing to runerpkg_db/pkginfo/, record intended file creations, overwrites, and deletions into a flat binary journal log.

Atomic File Swapping: Never write directly to target system paths during installation. Extract payloads into a staging subdirectory (/var/lib/runepkg_dir/staging_<pid>/), then perform atomic POSIX rename() operations or transactional hardlink swaps during the COMMITTING phase.

Safe Abort Trap: Register system signal handlers (SIGINT, SIGTERM, SIGSEGV) to intercept abrupt termination. If caught outside the COMMITTING phase, the signal handler automatically transitions the FSM into RUNEPKG_STATE_ROLLBACK to purge temp files and exit with a non-zero status code, leaving the host untouched.

3. FSM Transition Engine Core Loop
Implement a state-dispatch loop that strictly governs execution flow and handles error propagation:

C

RunepkgState runepkg_execute_transaction(TransactionContext *ctx) {
    RunepkgState current_state = RUNEPKG_STATE_PREPARING;
    
    while (current_state != RUNEPKG_STATE_IDLE && 
           current_state != RUNEPKG_STATE_FAILED) {
        
        switch (current_state) {
            case RUNEPKG_STATE_PREPARING:
                current_state = step_prepare(ctx) == 0 ? 
                                RUNEPKG_STATE_VALIDATING : RUNEPKG_STATE_ROLLBACK;
                break;
                
            case RUNEPKG_STATE_VALIDATING:
                current_state = step_validate(ctx) == 0 ? 
                                RUNEPKG_STATE_STAGING : RUNEPKG_STATE_ROLLBACK;
                break;
                
            case RUNEPKG_STATE_STAGING:
                current_state = step_stage(ctx) == 0 ? 
                                RUNEPKG_STATE_COMMITTING : RUNEPKG_STATE_ROLLBACK;
                break;
                
            case RUNEPKG_STATE_COMMITTING:
                if (step_commit(ctx) != 0) {
                    current_state = RUNEPKG_STATE_ROLLBACK;
                } else {
                    current_state = RUNEPKG_STATE_CLEANUP;
                }
                break;
                
            case RUNEPKG_STATE_ROLLBACK:
                step_rollback(ctx);
                current_state = RUNEPKG_STATE_FAILED;
                break;
                
            case RUNEPKG_STATE_CLEANUP:
                step_cleanup(ctx);
                current_state = RUNEPKG_STATE_IDLE;
                break;
                
            default:
                current_state = RUNEPKG_STATE_FAILED;
                break;
        }
    }
    return current_state;
}

An RAII (Resource Acquisition Is Initialization) transaction guard bridges this gap seamlessly:

// runepkg_guard.hpp
#pragma once

extern "C" {
    #include "runepkg_state.h"
}

class RunepkgTransactionGuard {
private:
    TransactionContext* ctx_;
    bool committed_;

public:
    explicit RunepkgTransactionGuard(TransactionContext* ctx) : ctx_(ctx), committed_(false) {
        // Initialize state to PREPARING
        ctx_->state = RUNEPKG_STATE_PREPARING;
    }

    ~RunepkgTransactionGuard() {
        if (!committed_ && ctx_->state != RUNEPKG_STATE_IDLE) {
            // Stack unwound due to an exception or early exit without commit!
            ctx_->state = RUNEPKG_STATE_ROLLBACK;
            step_rollback(ctx_);
            step_cleanup(ctx_);
        }
    }

    void commit() {
        committed_ = true;
    }
    
    // Delete copy semantics to prevent double-free or dual-rollback triggers
    RunepkgTransactionGuard(const RunepkgTransactionGuard&) = delete;
    RunepkgTransactionGuard& operator=(const RunepkgTransactionGuard&) = delete;
};

Step 1: Low-Level Logger Core (runepkg_util.c / runepkg_state.c)
Implement the base file I/O, timestamp generation, and formatting functions in C. Because the C core owns the state machine, it initializes the transaction log file handle when a command starts.

#include <stdio.h>
#include <stdarg.h>
#include <time.h>
#include <sys/stat.h>
#include <string.h>

static FILE *current_tx_log = NULL;
static char current_tx_path[512] = {0};

int runepkg_log_init(const char *log_dir) {
    char timestamp[64];
    time_t now = time(NULL);
    struct tm *t = localtime(&now);
    strftime(timestamp, sizeof(timestamp), "%Y%m%d-%H%M%S", t);

    // Ensure log directory exists (using your utility/FHS checks)
    snprintf(current_tx_path, sizeof(current_tx_path), "%s/transaction-%s.log", log_dir, timestamp);
    
    current_tx_log = fopen(current_tx_path, "w");
    if (!current_tx_log) return -1;

    fprintf(current_tx_log, "[INIT] Transaction started at %s\n", timestamp);
    fflush(current_tx_log);
    return 0;
}

void runepkg_log_write(const char *level, const char *fmt, ...) {
    if (!current_tx_log) return;

    va_list args;
    va_start(args, fmt);
    fprintf(current_tx_log, "[%s] ", level);
    vfprintf(current_tx_log, fmt, args);
    fprintf(current_tx_log, "\n");
    fflush(current_tx_log);
    va_end(args);
}

void runepkg_log_fail(const char *err_msg, const char *log_dir) {
    if (current_tx_log) {
        fprintf(current_tx_log, "[FATAL] %s\n", err_msg);
        fclose(current_tx_log);
        current_tx_log = NULL;
    }

    // Write / append to transaction_failure.log
    char fail_path[512];
    snprintf(fail_path, sizeof(fail_path), "%s/transaction_failure.log", log_dir);
    FILE *fail_log = fopen(fail_path, "a");
    if (fail_log) {
        time_t now = time(NULL);
        fprintf(fail_log, "[%ld] FAILED: %s (Detailed trace in %s)\n", (long)now, err_msg, current_tx_path);
        fclose(fail_log);
    }
}

void runepkg_log_close(int clean_up_enabled, const char *log_dir) {
    if (current_tx_log) {
        fclose(current_tx_log);
        current_tx_log = NULL;
    }
    if (clean_up_enabled && current_tx_path[0] != '\0') {
        remove(current_tx_path); // Delete timestamped log on success if clean_up=yes
    }
}
