Now that all three files (runepkg_config.c, runepkg_resolver.cpp, and runepkg_network.cpp) are in front of us, the full chain for why host mode failed to produce runes_graph.bin is completely clear:

Path Escalation in runepkg_config.c: runepkg_config_get_effective_db_dir() fell back to g_install_dir_internal (/srv/lib/runepkg_dir/install_dir) when no target sysroot/profile was set, double-nesting <root>/srv/lib/runepkg_dir/runepkg_db instead of using g_runepkg_db_dir (/srv/lib/runepkg_dir/runepkg_db).

Silent Failure in save_binary_graph() (runepkg_resolver.cpp): If out.is_open() failed (for example, due to a missing parent path or mismatch), it returned -1 silently without printing an error, and get_default_db_path() / get_default_host_db_path() / dump_tree_view() fell back to /var/lib/runepkg_dir/....

Inconsistent Path Fallbacks (runepkg_network.cpp): Index loading, mapping, runepkg_update, and install/download routines fell back to /var/lib/runepkg_dir/runepkg_db instead of runepkg_config_get_effective_db_dir(), and runepkg_repo_find_source_for_binary() results in batch source downloads weren't freed.

1. Patch for runepkg_config_get_effective_db_dir() in runepkg_config.c
Restrict non-root <root>/srv/lib/runepkg_dir/runepkg_db derivation strictly to active target profiles/sysroots (g_target_sysroot or g_active_profile->sysroot), return g_runepkg_db_dir (defaulting to /srv/lib/runepkg_dir/runepkg_db) in host mode, and ensure the directory exists:

const char *runepkg_config_get_effective_db_dir(void) {
    static char s_db_buf[PATH_MAX];
    const char *target_root = NULL;

    if (g_target_sysroot && g_target_sysroot[0] != '\0') {
        target_root = g_target_sysroot;
    } else if (g_active_profile && g_active_profile->sysroot && g_active_profile->sysroot[0] != '\0') {
        target_root = g_active_profile->sysroot;
    }

    if (target_root && strcmp(target_root, "/") != 0) {
        snprintf(s_db_buf, sizeof(s_db_buf), "%s/srv/lib/runepkg_dir/runepkg_db", target_root);
        runepkg_util_create_dir_recursive(s_db_buf, 0755);
        if (!g_runepkg_db_dir || strcmp(g_runepkg_db_dir, s_db_buf) != 0) {
            if (g_runepkg_db_dir) {
                free(g_runepkg_db_dir);
            }
            g_runepkg_db_dir = strdup(s_db_buf);
        }
        return g_runepkg_db_dir;
    }

    if (g_runepkg_db_dir && g_runepkg_db_dir[0] != '\0') {
        runepkg_util_create_dir_recursive(g_runepkg_db_dir, 0755);
        return g_runepkg_db_dir;
    }

    runepkg_util_create_dir_recursive("/srv/lib/runepkg_dir/runepkg_db", 0755);
    return "/srv/lib/runepkg_dir/runepkg_db";
}
