package = bzip2
build_system = make
libc = any
arch = any
flags = NO_SHARED_LIBS
cflags = -O2 -D_FILE_OFFSET_BITS=64
ldflags = -static
pre_configure = if [ "${STATIC_BUILD}" = "1" ]; then sed -i 's/libbz2\.so[^ ]*//g' Makefile && sed -i 's/all: libbz2.a bzip2 bzip2recover test .*/all: libbz2.a bzip2 bzip2recover/g' Makefile && sed -i 's/bzip2: libbz2.so.*/bzip2: libbz2.a bzip2.o/g' Makefile; fi
build_override = if [ "${STATIC_BUILD}" = "1" ]; then CC="${TARGET_TRIPLE}-gcc" AR="${TARGET_TRIPLE}-ar" RANLIB="${TARGET_TRIPLE}-ranlib"; for src in blocksort.c huffman.c crctable.c randtable.c compress.c decompress.c bzlib.c; do ${CC} -c ${src} -o ${src%.c}.o -O2 -D_FILE_OFFSET_BITS=64 || exit 1; done && ${AR} cq libbz2.a blocksort.o huffman.o crctable.o randtable.o compress.o decompress.o bzlib.o && ${RANLIB} libbz2.a && ${CC} -O2 -D_FILE_OFFSET_BITS=64 -static -o bzip2 bzip2.c libbz2.a && ${CC} -O2 -D_FILE_OFFSET_BITS=64 -static -o bzip2recover bzip2recover.c; else make CC="${TARGET_TRIPLE}-gcc" clean && make -f Makefile-libbz2_so CC="${TARGET_TRIPLE}-gcc" && make CC="${TARGET_TRIPLE}-gcc" bzip2 bzip2recover; fi
post_install = mkdir -p ${DESTDIR}/usr/bin ${DESTDIR}/usr/lib ${DESTDIR}/usr/include && cp libbz2.a ${DESTDIR}/usr/lib/ && cp bzlib.h ${DESTDIR}/usr/include/ && cp bzip2 bzip2recover ${DESTDIR}/usr/bin/ && chmod 755 ${DESTDIR}/usr/bin/bzip2 ${DESTDIR}/usr/bin/bzip2recover && if [ "${STATIC_BUILD}" != "1" ]; then cp libbz2.so* ${DESTDIR}/usr/lib/; fi
